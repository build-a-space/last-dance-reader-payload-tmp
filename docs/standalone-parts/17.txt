); els.tocClose.addEventListener('click', closeToc); els.tocBackdrop.addEventListener('click', e => { if (e.target === els.tocBackdrop) closeToc(); });

      window.addEventListener('keydown', e => {
        if (!els.tocBackdrop.hidden) { if (e.key === 'Escape') closeToc(); return; }
        if (['INPUT', 'TEXTAREA', 'BUTTON'].includes(document.activeElement?.tagName) && e.key !== 'Escape') return;
        if (e.key === 'ArrowRight' || e.key === 'PageDown') { e.preventDefault(); goTo(pageIndex + 1); }
        else if (e.key === 'ArrowLeft' || e.key === 'PageUp') { e.preventDefault(); goTo(pageIndex - 1); }
        else if (e.key === 'Home') { e.preventDefault(); goTo(0); }
        else if (e.key === 'End' && book) { e.preventDefault(); goTo(book.chapters.length); }
      });

      els.stage.addEventListener('touchend', e => { if (!touchStart) return; const t = e.changedTouches[0], dx = t.clientX - touchStart.x, dy = t.clientY - touchStart.y; touchStart = null; if (Math.abs(dx) > 60 && Math.abs(dx) > Math.abs(dy)) goTo(pageIndex + (dx < 0 ? 1 : -1)); }, { passive: true });
